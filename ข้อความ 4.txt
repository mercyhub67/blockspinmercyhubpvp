local function GetDistanceFromStart(Objective, Start)
	if typeof(Objective) == "Vector3" and typeof(Start) == "Vector3" then
		return (Objective - Start).Magnitude
	elseif typeof(Objective) == "Vector2" and typeof(Start) == "Vector2" then
		return (Objective - Start).Magnitude
	else
		return math.huge
	end
end

local function isBehindWall(startPos, endPos)
	if not startPos or not endPos then return false end
	local ray = Ray.new(startPos, endPos - startPos)
	local ignoreList = {}
	if Character then table.insert(ignoreList, Character) end
	local targets = c().TargetSlientAim
	if targets and targets.Character then table.insert(ignoreList, targets.Character) end
	local hit = workspace:FindPartOnRayWithIgnoreList(ray, ignoreList)
	return hit ~= nil
end

function GetInfo(w)

local function Teleport(destination, value, t)
	c().StopWalking = false 
	c()['Running'] = true 
	local char = Client.Character or Client.CharacterAdded:Wait()
	local RootPart = char:WaitForChild("HumanoidRootPart")
	local Humanoid = char:WaitForChild("Humanoid")

	if Hee then 
		Hee = nil
	end

	local path = PathfindingService:CreatePath({
		AgentCanJump = true,
		AgentJumpHeight = 2.5,
		AgentHeight = 4.5,
		AgentRadius = 2.5,
		AgentMaxSlope = 90,
		Costs = {
			BlockedNode = 50,
			DoorArea = 1,
		},
	})

	Hee = path 

	local success, err = pcall(function()
		path:ComputeAsync(RootPart.Position, destination)
	end)

	if not success then
		return
	end

	if path.Status == Enum.PathStatus.Success then
		local waypoints = path:GetWaypoints()
		for _, wp in getn(waypoints) do
			local offsetY = (wp.Action == Enum.PathWaypointAction.Jump) and 10 or 3
			local targetPos = wp.Position + Vector_new(0, offsetY, 0)
			local startPos = RootPart.Position
			local dir = (targetPos - startPos).Unit
			local dist = (targetPos - startPos).Magnitude
			local movedDist = 0
			local speed = c().InstantTeleportSpeed or 34
			local startTime = tick()

			while movedDist < dist and not c().StopWalking and value do waits()
				c()['Running'] = true 
				if c().StopWalking then break end
				if not value then c()['Running'] = false break end
				if Humanoid.Health <= 0 then c()['Running'] = false break end
				if Detect() then c()['Running'] = false Teleport(destination, value, t) return end
				if t and GetAtmIsNotHacked(t) == false then c()['Running'] = false return end 
				if t then if CheckingIsMinigame() then c()['Running'] = false Teleport(destination, value, t) return end end
				local elapsedTime = tick() - startTime
				movedDist = math.min(elapsedTime * speed, dist)
				local newPos = startPos + dir * movedDist
				if Humanoid.Sit then
					Humanoid.Sit = false
				end
				RootPart:PivotTo(CFrame.new(newPos))
				Ac("set_sprinting_1", true)
			end
			c()['Running'] = false
		end
	else
		Hee = nil
		c()['Running'] = false
		-- Teleport(destination + Vector3.new(-0.5,0,-0.5), value, t) 
		-- return
	end
	Hee = nil 
end;

local function Drive(model, destination, value, t)
	if not model or not model.PrimaryPart then
		c()['Running'] = false
		return
	end

	c()["StopWalking"] = false
	local path = PathfindingService:CreatePath({
		AgentRadius = c().VechineType == "car" and 5 or 2.8,
		AgentHeight = 5,
		AgentCanJump = true,
		AgentMaxSlope = 45,
		Costs = {
			BlockedNode = 50,
			Cars = 1,
		},
	})

	local success, err = pcall(function()
		path:ComputeAsync(model.PrimaryPart.Position, destination)
	end)

	if not success or path.Status ~= Enum.PathStatus.Success then
		c()['Running'] = false
		return
	end

	local waypoints = path:GetWaypoints()
	for i, wp in getn(waypoints) do
		if not model or not model.PrimaryPart then
			c()['Running'] = false
			return
		end

		local startPos = model:GetPivot().Position
		local goalPos = wp.Position + Vector_new(0, 2, 0)
		local dir = (goalPos - startPos).Unit
		local dist = (goalPos - startPos).Magnitude
		local movedDist = 0
		local speed = c().VechineSpeed or 55
		local startTime = tick()

		while movedDist < dist and value do 
			waits()
			c()['Running'] = true 

			if not model or not model.PrimaryPart then
				c()['Running'] = false
				return
			end

			if Detect() then
				c()['Running'] = false
				Drive(model, destination, value, t)
				return
			end

			if not Humanoid.Sit then 
				c()['Running'] = false
				Drive(model, destination, value, t)
				return
			end

			if CheckingIsMinigame() or (t and GetAtmIsNotHacked(t) == false) or not value then
				c()['Running'] = false 
				return
			end

			for _, v in get(model:GetDescendants()) do
				if v:IsA("BasePart") and not v:FindFirstChild("PathfindingModifier") then
					local modifier = Instance.new("PathfindingModifier")
					modifier.Parent = v
					modifier.PassThrough = true
					modifier.Label = "Cars"
				end
			end

			local dt = tick() - startTime
			movedDist = math.min(dt * speed, dist)
			local newPos = startPos + dir * movedDist

			if wp.Action == Enum.PathWaypointAction.Jump then
				newPos = newPos + Vector_new(0, 11, 0)
			end

			local Rotation = CFrame.lookAt(newPos, newPos + dir)

			model:PivotTo(Rotation)

			for _, part in get(model:GetDescendants()) do
				if part:IsA("BasePart") then
					part.Velocity = Vector3.zero
					part.RotVelocity = Vector3.zero
					part.AssemblyLinearVelocity = Vector3.zero
					part.AssemblyAngularVelocity = Vector3.zero
				end
			end
		end
		for _, part in get(model:GetDescendants()) do
			if part:IsA("BasePart") then
				part.Velocity = Vector3.zero
				part.RotVelocity = Vector3.zero
				part.AssemblyLinearVelocity = Vector3.zero
				part.AssemblyAngularVelocity = Vector3.zero
			end
		end
		c()['Running'] = false 
	end
end;

local function MoveWithCondition(target,value,config)
	if c().EnabledVechine then 
		Get_Vehicles(GetCarFromType(c().SelectVechine or "bike"),function(v)
			if not Detect() and not CheckingIsMinigame() then 
				if not Humanoid.Sit then 
					local dist = GetDistanceFromRootPart(v.PrimaryPart)
					if dist < 5 then 
						for _,b in ipairs(v.PrimaryPart:GetDescendants()) do 
							if b:IsA('ProximityPrompt') then fireproximityprompt(b) end
						end
					else
						v:PivotTo(RootPart.CFrame * CFrame_new(0,3,-2))
					end
				else
					local t = config and config.T and config.T.Parent
					if t then
						Drive(v, target.Position, value, t)
					end
				end
			end
		end) 
	else
		local t = config and config.T and config.T.Parent or target
		Teleport(target.Position,value,t)
	end
end

local function GetNearThing(folder, path, condition)
	local closest, shortestDist = nil, math.huge
	if not folder then return nil end

	for _, obj in getn(folder:GetChildren()) do
		if string_find(obj.Name,path) then
			if not condition or condition(obj) then
				if obj:IsA("BasePart") then
					local dist = GetDistanceFromRootPart(obj.CFrame)
					if dist < shortestDist then
						closest = obj
						shortestDist = dist
					end
				elseif obj:IsA('Model') then
					local dist = GetDistanceFromRootPart(obj.Area)
					if dist < shortestDist then
						closest = obj
						shortestDist = dist
					end
				end 
			end
		end
	end

	return closest
end;

local function Tg(def,Tab,Title,Keys,keybind)
	local taskThread
	local toggle = Tab:Toggle({
		Title = Title,
		Desc = "Fucking dont read this",
		Value = def or c()[Keys] or Settings[Keys],
		Callback = function(state)
			Settings[Keys] = state
			c()[Keys] = state
			SC()
			if not state then 
				if taskThread then
					cancel(taskThread)
					taskThread = nil
				end
			end

			if state then 
				if func[Keys] then
					taskThread = task.spawn(func[Keys])
				end
			end
		end
	})
	if toggle then toggle.Callback(c()[Keys]) end 
	if keybind then 
		Tab:Keybind({
			Flag = "Keybind",
			Title = "Keybind",
			Desc = "Just a keybind",
			Value = keybind,
			Callback = function(v)
				toggle:Set(not c()[Keys])
			end
		})
	end
	return toggle 
end

local function Slider(Tab,Title,Steps,Mins,Maxs,Def,Keys)
	local slider = Tab:Slider({
		Flag = Title,
		Title = Title,
		Step = Steps,
		Value = {
			Min = Mins,
			Max = Maxs,
			Default = Settings[Keys] or Def,
		},
		Callback = function(value)
			Settings[Keys] = value 
			c()[Keys] = value 
			SC()
		end
	})
	if slider then slider.Callback(c()[Keys]) end 
	return slider
end

local function Input(Tab, Title, Keys)
	local input = Tab:Input({
		Title = Title,
		Type = "Input",
		Placeholder = "Enter here...",
		Callback = function(value)
			Settings[Keys] = value
			c()[Keys] = value
			SC()
		end
	})
	if input and input.Callback then input.Callback(c()[Keys]) end
	return input
end

local function Dropdown(Tab, Title, Keys, Values, Multi, calls)
	local dropdown = Tab:Dropdown({
		Title = Title,
		Values = Values or {"nigger","Back"},
		Multi = Multi or false,
		AllowNone = Multi or false, 
		Value = c()[Keys] or Settings[Keys] or Values[1], 
		Callback = function(option)
			Settings[Keys] = option
			c()[Keys] = option
			if calls then calls(option) end 
			SC()
		end
	})

	if dropdown and dropdown.Callback then
		dropdown.Callback(c()[Keys] or (Multi and {} or Values[1]))
	end

	return dropdown
end

local function Colorpicker(Tab, Title,Desc, Keys, Default)

		colorpicker.Callback(c()[Keys] or Default or Color3.fromRGB(255,255,255))
	end

	return colorpicker
end

local oreq = syn and syn.request or http_request or request

local function Req(op)
	local safeOptions = {}
	for k, v in pairs(op) do
		safeOptions[k] = v
	end

	if safeOptions.Url then
		local randomParam = tostring(math.random(100000, 999999))
		local separator = safeOptions.Url:find('?') and '&' or '?'
		safeOptions.Url = safeOptions.Url .. separator .. '_t=' .. randomParam
	end

	if not safeOptions.Headers then
		safeOptions.Headers = {}
	end
	safeOptions.Headers['User-Agent'] = 'Roblox/WinInet'
	safeOptions.Headers['Cache-Control'] = 'no-cache'

	local success, result = pcall(function()
		return oreq(safeOptions)
	end)

	if success then
		return result
	else
		warn(result)
	end
end
